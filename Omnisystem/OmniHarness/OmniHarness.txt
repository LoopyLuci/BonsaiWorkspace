## 1. Architecture Overview

The OmniHarness is split into three core tiers:

- **Rust Kernel** – Immutable event store, WASM sandbox, gRPC server, mesh networking. This is the hardened foundation that never corrupts.
- **Clojure Orchestrator** – The “mind” with a ReAct loop, HTN planner, policy engine, and self‑evolution logic. Communicates with the kernel via gRPC.
- **ClojureScript GUI** – A web interface that displays agent thoughts, lets the user switch AI models, and visualises the memory graph.

All components share a single `.proto` file that defines the **Century Protocol**.

```rust
use serde::{Deserialize, Serialize};
use sha2::{Digest, Sha256};
use std::io::{Error, ErrorKind};
use std::path::PathBuf;
use std::sync::Arc;
use tokio::fs::{self, File, OpenOptions};
use tokio::io::{AsyncBufReadExt, AsyncWriteExt, BufReader};
use tokio::sync::RwLock;

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub struct SystemEvent {
    pub id: String,
    pub timestamp_utc: i64,
    pub module_source: String,
    pub event_type: String,
    pub payload_json: String,
    pub previous_hash: String,
    pub current_hash: String,
}

impl SystemEvent {
    pub fn calculate_hash(
        timestamp: i64,
        module_source: &str,
        event_type: &str,
        payload_json: &str,
        previous_hash: &str,
    ) -> String {
        let mut hasher = Sha256::new();
        hasher.update(timestamp.to_string().as_bytes());
        hasher.update(module_source.as_bytes());
        hasher.update(event_type.as_bytes());
        hasher.update(payload_json.as_bytes());
        hasher.update(previous_hash.as_bytes());
        format!("{:x}", hasher.finalize())
    }
}

pub struct PersistentEventStore {
    storage_path: PathBuf,
    latest_hash: Arc<RwLock<String>>,
}

impl PersistentEventStore {
    pub async fn new(path: &str) -> Result<Self, Error> {
        let storage_path = PathBuf::from(path);
        if let Some(parent) = storage_path.parent() {
            fs::create_dir_all(parent).await?;
        }

        let latest_hash = if storage_path.exists() {
            Self::recover_and_verify(&storage_path).await?
        } else {
            String::from("e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855")
        };

        Ok(Self {
            storage_path,
            latest_hash: Arc::new(RwLock::new(latest_hash)),
        })
    }

    pub async fn append_event(
        &self,
        module_source: &str,
        event_type: &str,
        payload_json: &str,
    ) -> Result<SystemEvent, Error> {
        let mut tip = self.latest_hash.write().await;

        let timestamp = chrono::Utc::now().timestamp();
        let new_hash = SystemEvent::calculate_hash(
            timestamp,
            module_source,
            event_type,
            payload_json,
            &tip,
        );

        let event = SystemEvent {
            id: uuid::Uuid::new_v4().to_string(),
            timestamp_utc: timestamp,
            module_source: module_source.to_string(),
            event_type: event_type.to_string(),
            payload_json: payload_json.to_string(),
            previous_hash: tip.clone(),
            current_hash: new_hash.clone(),
        };

        let line = serde_json::to_string(&event)? + "\n";
        let mut file = OpenOptions::new()
            .create(true)
            .append(true)
            .open(&self.storage_path)
            .await?;
        file.write_all(line.as_bytes()).await?;
        file.sync_data().await?;

        *tip = new_hash;
        Ok(event)
    }

    pub async fn append_external_event(&self, external_event: SystemEvent) -> Result<(), Error> {
        let mut tip = self.latest_hash.write().await;

        if external_event.previous_hash != *tip {
            return Err(Error::new(
                ErrorKind::InvalidData,
                format!("Mesh fork detected. Expected tip {} but event has previous_hash {}", *tip, external_event.previous_hash),
            ));
        }

        let verified_hash = SystemEvent::calculate_hash(
            external_event.timestamp_utc,
            &external_event.module_source,
            &external_event.event_type,
            &external_event.payload_json,
            &external_event.previous_hash,
        );

        if verified_hash != external_event.current_hash {
            return Err(Error::new(
                ErrorKind::InvalidData,
                "Mesh event signature verification failed.",
            ));
        }

        let line = serde_json::to_string(&external_event)? + "\n";
        let mut file = OpenOptions::new()
            .create(true)
            .append(true)
            .open(&self.storage_path)
            .await?;
        file.write_all(line.as_bytes()).await?;
        file.sync_data().await?;

        *tip = external_event.current_hash;
        Ok(())
    }

    pub async fn verify_chain(&self) -> Result<bool, Error> {
        if !self.storage_path.exists() {
            return Ok(true);
        }
        let file = File::open(&self.storage_path).await?;
        let mut reader = BufReader::new(file).lines();
        let mut expected_prev =
            String::from("e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855");

        while let Some(line) = reader.next_line().await? {
            let event: SystemEvent = serde_json::from_str(&line)?;
            if event.previous_hash != expected_prev {
                return Err(Error::new(
                    ErrorKind::InvalidData,
                    format!("Chain broken at event {}", event.id),
                ));
            }
            let recalc = SystemEvent::calculate_hash(
                event.timestamp_utc,
                &event.module_source,
                &event.event_type,
                &event.payload_json,
                &event.previous_hash,
            );
            if event.current_hash != recalc {
                return Err(Error::new(
                    ErrorKind::InvalidData,
                    format!("Hash mismatch at event {}", event.id),
                ));
            }
            expected_prev = event.current_hash;
        }
        Ok(true)
    }

    pub async fn current_tip(&self) -> String {
        self.latest_hash.read().await.clone()
    }

    async fn recover_and_verify(path: &PathBuf) -> Result<String, Error> {
        let file = File::open(path).await?;
        let reader = BufReader::new(file).lines();
        let mut expected_prev =
            String::from("e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855");
        let mut last_hash = expected_prev.clone();

        let mut lines = reader;
        while let Some(line) = lines.next_line().await? {
            let event: SystemEvent = serde_json::from_str(&line)?;
            if event.previous_hash != expected_prev {
                return Err(Error::new(
                    ErrorKind::InvalidData,
                    format!("Chain broken at event {}", event.id),
                ));
            }
            let recalc = SystemEvent::calculate_hash(
                event.timestamp_utc,
                &event.module_source,
                &event.event_type,
                &event.payload_json,
                &event.previous_hash,
            );
            if event.current_hash != recalc {
                return Err(Error::new(
                    ErrorKind::InvalidData,
                    format!("Hash mismatch at event {}", event.id),
                ));
            }
            expected_prev = event.current_hash.clone();
            last_hash = event.current_hash;
        }
        Ok(last_hash)
    }
}
```

### 3.7 `rust-kernel/src/kernel/sandbox.rs`

```rust
use anyhow::{anyhow, Result};
use wasmtime::*;
use wasmtime_wasi::sync::WasiCtxBuilder;

pub struct Sandbox {
    engine: Engine,
    linker: Linker<()>,
}

impl Sandbox {
    pub fn new() -> Result<Self> {
        let mut config = Config::new();
        config.consume_fuel(true);
        config.max_wasm_memory(64 * 1024 * 1024);
        let engine = Engine::new(&config)?;
        let mut linker = Linker::new(&engine);
        wasmtime_wasi::add_to_linker(&mut linker, |s| s)?;
        Ok(Self { engine, linker })
    }

    pub fn run(&self, wasm_bytes: &[u8], args: Vec<String>, fuel: u64) -> Result<String> {
        let mut store = Store::new(&self.engine, ());
        store.set_fuel(fuel)?;

        let wasi = WasiCtxBuilder::new()
            .args(&args)
            .inherit_stdout()
            .build();
        *store.data_mut() = wasi;

        let module = Module::new(&self.engine, wasm_bytes)?;
        let instance = self.linker.instantiate(&mut store, &module)?;
        let start = instance.get_typed_func::<(), ()>(&mut store, "_start")?;

        match start.call(&mut store, ()) {
            Ok(_) => {
                let fuel_consumed = fuel - store.get_fuel();
                Ok(format!("SUCCESS (fuel used: {})", fuel_consumed))
            }
            Err(e) => Err(anyhow!("Sandbox trap: {}", e)),
        }
    }
}
```

### 3.8 `rust-kernel/src/kernel/bootloader.rs`

```rust
use crate::kernel::event_store::{PersistentEventStore, SystemEvent};
use crate::kernel::sandbox::Sandbox;
use crate::server::grpc;
use crate::mesh::node::MeshNode;
use std::sync::Arc;
use tokio::signal;
use tokio::sync::mpsc;
use anyhow::Result;

pub struct GenesisBootloader {
    event_store: Arc<PersistentEventStore>,
    _sandbox: Arc<Sandbox>,
}

impl GenesisBootloader {
    pub async fn initialize(storage_path: &str) -> Result<Self> {
        println!("[BOOT] Initializing Century OmniHarness Kernel...");

        let store = PersistentEventStore::new(storage_path).await
            .map_err(|e| anyhow::anyhow!("Event Store init failed: {}", e))?;
        let event_store = Arc::new(store);

        println!("[BOOT] Verifying cryptographic Merkle chain...");
        event_store.verify_chain().await
            .map_err(|e| anyhow::anyhow!("Merkle chain corruption detected: {}", e))?;
        println!("[BOOT] Merkle chain verified. Tip hash: {}", event_store.current_tip().await);

        let sandbox_env = Sandbox::new()?;
        let _sandbox = Arc::new(sandbox_env);
        println!("[BOOT] WASM Sandbox environment primed.");

        Ok(Self {
            event_store,
            _sandbox,
        })
    }

    pub async fn run(self) -> Result<()> {
        let grpc_store = Arc::clone(&self.event_store);

        // 1. Spawn gRPC Server
        let grpc_handle = tokio::spawn(async move {
            if let Err(e) = grpc::serve(grpc_store).await {
                eprintln!("[FATAL] gRPC Server failed: {}", e);
            }
        });

        // 2. Initialize Mesh & Internal Event Router
        let (mesh_tx, mut mesh_rx) = mpsc::channel::<Vec<u8>>(1024);
        let mesh_node = MeshNode::new(mesh_tx)
            .map_err(|e| anyhow::anyhow!("Failed to initialize Mesh network: {}", e))?;

        let mesh_handle = tokio::spawn(async move {
            mesh_node.run().await;
        });

        // 3. Spawn Mesh Event Consumer
        let mesh_store = Arc::clone(&self.event_store);
        let consumer_handle = tokio::spawn(async move {
            while let Some(data) = mesh_rx.recv().await {
                if let Ok(external_event) = serde_json::from_slice::<SystemEvent>(&data) {
                    match mesh_store.append_external_event(external_event.clone()).await {
                        Ok(_) => println!("[MESH] Replicated external event ID: {}", external_event.id),
                        Err(e) => eprintln!("[MESH REJECTED] {}", e),
                    }
                }
            }
        });

        // Await Shutdown
        match signal::ctrl_c().await {
            Ok(()) => println!("\n[SHUTDOWN] Signal received. Executing graceful shutdown sequence..."),
            Err(err) => eprintln!("[ERROR] Unable to listen for shutdown signal: {}", err),
        }

        grpc_handle.abort();
        mesh_handle.abort();
        consumer_handle.abort();

        println!("[SHUTDOWN] Century Protocol boundary closed. Storage locks released. Goodbye.");
        Ok(())
    }
}
```

```rust
use tonic::{transport::Server, Request, Response, Status};
use std::sync::Arc;
use crate::kernel::event_store::PersistentEventStore;

pub mod proto {
    tonic::include_proto!("omniharness.v1");
}
use proto::event_store_service_server::{EventStoreService, EventStoreServiceServer};
use proto::{AppendRequest, AppendResponse, VerifyRequest, VerifyResponse};

pub struct EventStoreRpc {
    store: Arc<PersistentEventStore>,
}

#[tonic::async_trait]
impl EventStoreService for EventStoreRpc {
    async fn append_event(&self, req: Request<AppendRequest>) -> Result<Response<AppendResponse>, Status> {
        let r = req.into_inner();
        match self.store.append_event(&r.module_source, &r.event_type, &r.payload_json).await {
            Ok(ev) => Ok(Response::new(AppendResponse {
                event_hash: ev.current_hash,
                success: true,
            })),
            Err(e) => Err(Status::internal(format!("Persistence failure: {}", e))),
        }
    }

    async fn verify_chain(&self, _: Request<VerifyRequest>) -> Result<Response<VerifyResponse>, Status> {
        match self.store.verify_chain().await {
            Ok(valid) => Ok(Response::new(VerifyResponse {
                is_valid: valid,
                tip_hash: self.store.current_tip().await,
            })),
            Err(e) => Err(Status::internal(format!("Verification error: {}", e))),
        }
    }
}

pub async fn serve(store: Arc<PersistentEventStore>) -> Result<(), Box<dyn std::error::Error>> {
    let addr = "[::1]:50051".parse()?;
    let svc = EventStoreRpc { store };
    println!("Century Protocol gRPC server listening on {}", addr);
    Server::builder()
        .add_service(EventStoreServiceServer::new(svc))
        .serve(addr)
        .await?;
    Ok(())
}
```

### 3.12 `rust-kernel/src/mesh/node.rs`

```rust
use libp2p::{
    core::upgrade,
    gossipsub::{self, IdentTopic as GossipsubTopic, MessageAuthenticity},
    identify, mdns, noise,
    swarm::{NetworkBehaviour, SwarmEvent},
    tcp, yamux, PeerId, Swarm, SwarmBuilder,
};
use std::error::Error;
use std::time::Duration;
use tokio::sync::mpsc;
use libp2p::identity::Keypair;

#[derive(NetworkBehaviour)]
pub struct MeshBehaviour {
    pub gossipsub: gossipsub::Behaviour,
    pub mdns: mdns::tokio::Behaviour,
    pub identify: identify::Behaviour,
}

pub struct MeshNode {
    swarm: Swarm<MeshBehaviour>,
    event_sender: mpsc::Sender<Vec<u8>>,
    topic: GossipsubTopic,
}

impl MeshNode {
    pub fn new(event_tx: mpsc::Sender<Vec<u8>>) -> Result<Self, Box<dyn Error>> {
        let local_key = Keypair::generate_ed25519();
        let peer_id = PeerId::from(local_key.public());
        println!("[MESH] Initializing Mesh Node. Local PeerID: {}", peer_id);

        let transport = tcp::tokio::Transport::default()
            .upgrade(upgrade::Version::V1)
            .authenticate(noise::Config::new(&local_key)?)
            .multiplex(yamux::Config::default())
            .boxed();

        let gossipsub_config = gossipsub::ConfigBuilder::default()
            .heartbeat_interval(Duration::from_secs(1))
            .validation_mode(gossipsub::ValidationMode::Strict)
            .build()?;

        let gossipsub = gossipsub::Behaviour::new(
            MessageAuthenticity::Signed(local_key.clone()),
            gossipsub_config,
        )?;

        let mdns = mdns::tokio::Behaviour::new(mdns::Config::default(), peer_id)?;
        let identify = identify::Behaviour::new(identify::Config::new(
            "omniharness/1.0".into(),
            local_key.public(),
        ));

        let behaviour = MeshBehaviour { gossipsub, mdns, identify };
        let swarm = SwarmBuilder::with_tokio_executor(transport, behaviour, peer_id).build();
        let topic = GossipsubTopic::new("omniharness-events");

        Ok(Self { swarm, event_sender: event_tx, topic })
    }

    pub async fn run(mut self) {
        if let Err(e) = self.swarm.behaviour_mut().gossipsub.subscribe(&self.topic) {
            eprintln!("[MESH ERROR] Failed to subscribe to gossipsub topic: {}", e);
            return;
        }

        if let Err(e) = self.swarm.listen_on("/ip4/0.0.0.0/tcp/0".parse().unwrap()) {
            eprintln!("[MESH ERROR] Failed to bind local TCP port: {}", e);
            return;
        }

        loop {
            tokio::select! {
                event = self.swarm.select_next_some() => match event {
                    SwarmEvent::Behaviour(MeshBehaviourEvent::Gossipsub(gossipsub::Event::Message { message, .. })) => {
                        if let Err(e) = self.event_sender.send(message.data).await {
                            eprintln!("[MESH ERROR] Internal channel routing failed: {}", e);
                        }
                    }
                    SwarmEvent::Behaviour(MeshBehaviourEvent::Mdns(mdns::Event::Discovered(list))) => {
                        for (peer_id, multiaddr) in list {
                            println!("[MESH] Discovered peer {} at {}", peer_id, multiaddr);
                            self.swarm.dial(peer_id).ok();
                        }
                    }
                    SwarmEvent::Behaviour(MeshBehaviourEvent::Mdns(mdns::Event::Expired(list))) => {
                        for (peer_id, _) in list {
                            println!("[MESH] Peer {} vanished from mDNS.", peer_id);
                        }
                    }
                    _ => {}
                }
            }
        }
    }
}
```

```rust
use std::fs;
use std::process::{Command, exit};
use std::path::Path;

fn main() {
    let state_path = "/var/lib/omniharness/patch-state.json";
    if !Path::new(state_path).exists() {
        launch_harness("current");
        return;
    }

    let state: serde_json::Value = serde_json::from_str(
        &fs::read_to_string(state_path).expect("Failed to read patch state")
    ).expect("Invalid patch state JSON");

    let staged_binary = state["staged_binary"].as_str().expect("Missing staged_binary");
    let snapshot_path = state["snapshot_path"].as_str().expect("Missing snapshot_path");

    // Atomic swap
    fs::rename("/opt/omniharness/bin/current", "/opt/omniharness/bin/previous").ok();
    fs::rename(staged_binary, "/opt/omniharness/bin/current").ok();

    // Launch new binary in verification mode
    let child = Command::new("/opt/omniharness/bin/current")
        .arg("--verify-patch")
        .arg(snapshot_path)
        .spawn()
        .expect("Failed to launch new harness");

    let status = child.wait().expect("Child process wasn't running");
    if !status.success() {
        eprintln!("Patch verification failed; rolling back.");
        fs::rename("/opt/omniharness/bin/previous", "/opt/omniharness/bin/current").ok();
        launch_harness("current");
    } else {
        // Success; the new harness will finalize and remove patch state
        println!("Patch applied successfully.");
    }
}

fn launch_harness(slot: &str) {
    let mut child = Command::new(format!("/opt/omniharness/bin/{}", slot))
        .spawn()
        .expect("Failed to launch harness");
    child.wait().ok();
    exit(0);
}
```

```clojure
         "Action: system.ping\n"

(defn parse-llm-output [text]
  (try
    (let [lines (clojure.string/split-lines text)
          thought (some #(second (re-find #"^Thought: (.*)" %)) lines)
          action (some #(second (re-find #"^Action: (.*)" %)) lines)
          action-input (some #(second (re-find #"^ActionInput: (.*)" %)) lines)]
      {:thought thought
       :action action
       :action-input (when action-input (json/parse-string action-input true))})
    (catch Exception _
      {:action "FinalAnswer" :action-input {:answer text}})))

(defn execute-tool-safely [tool-name args]
  ;; placeholder: in future, call sandbox via gRPC
  (str "Mock result of " tool-name " with args " args))

(defn run-loop [llm max-steps objective]
  (go
    (let [ctx (atom [])]
      (loop [step 0 status :running]
        (if (or (>= step max-steps) (= status :completed))
          {:history @ctx :status status}
          (let [context-str (clojure.string/join "\n" @ctx)
                llm-output (generate llm context-str objective)
                parsed (parse-llm-output llm-output)
                {:keys [thought action action-input]} parsed]
            (events/append-event! "react-loop" "LLMThought" {:step step :thought thought})
            (if (or (nil? action) (= action "FinalAnswer"))
              (do
                (events/append-event! "react-loop" "TaskComplete" {:answer (get-in action-input [:answer] "Done")})
                {:history @ctx :status :completed})
              (do
                (let [tool-result (execute-tool-safely action action-input)]
                  (events/append-event! "react-loop" "ToolExecuted" {:tool action :result tool-result})
                  (swap! ctx conj (str "Action: " action "\nResult: " tool-result))
                  (recur (inc step) :running))))))))))

(defstate react-engine
  :start nil
  :stop nil)
```

```clojure
(ns omniharness.planner
  (:require [mount.core :as mount]))

(def primitive-tools #{"system.ping" "file.read" "file.write"})

(defn primitive? [action]
  (contains? primitive-tools (:tool action)))

(defn decompose [goal]
  (case goal
    "greet" [{:tool "system.ping" :params {:host "localhost"}}]
    [{:tool "system.ping" :params {:host goal}}]))

(defn plan [goal]
  (let [stack (list {:goal goal})
        plan-atom (atom [])]
    (loop [stack stack]
      (if (empty? stack)
        @plan-atom
        (let [item (first stack)
              remaining (rest stack)]
          (if (primitive? item)
            (recur remaining (do (swap! plan-atom conj item) remaining))
            (let [subtasks (decompose (:goal item))]
              (recur (concat subtasks remaining) nil))))))))
```
